max(dot(normal, lightDir), 0.0) : 0.0; vec3 diffuse = diff * vec3(1.0, 0.9, 0.7) * color; // Luce calda giallognola // Specular (solo se la luce è accesa e riflessi abilitati) vec3 specular = vec3(0.0); if (u_lightOn && u_reflections) { float specularStrength = 0.5; vec3 viewDir = normalize(u_viewPos - v_position); vec3 reflectDir = reflect(-lightDir, normal); float spec = pow(max(dot(viewDir, reflectDir), 0.0), 32.0); specular = specularStrength * spec * vec3(1.0, 0.9, 0.7); } // Attenuazione della luce con la distanza (ombre) float attenuation = 1.0; if (u_shadows) { float distance = length(u_lightPos - v_position); attenuation = 1.0 / (1.0 + 0.01 * distance + 0.001 * distance * distance); } // Rendering avanzato (aggiunge effetti di penombra) if (u_advancedRendering) { // Effetto vignette (bordi scuri) vec2 center = vec2(0.5, 0.5); float dist = distance(v_texcoord, center); float vignette = smoothstep(0.5, 0.35, dist); // Applica vignette all'illuminazione attenuation *= vignette; } // Combina i risultati vec3 result = (ambient + (diffuse + specular) * attenuation); // Tonemapping semplice per evitare eccessiva luminosità result = result / (result + vec3(1.0)); gl_FragColor = vec4(result, 1.0); }
